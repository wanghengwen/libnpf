# libnpf's public interface: a program that knows only npf.h, linked with the
# shared library (two instances on loopback, no platform)
if(NPF_BUILD_LIB AND NOT WIN32)
  find_package(Threads REQUIRED)
  # sessions between two instances: not without peers (NPF_PEERS=OFF)
  if(NPF_HAS_PEERS)
    add_executable(npf_api_test test_api.c)
    target_link_libraries(npf_api_test PRIVATE npf_shared Threads::Threads)
    npf_target_options(npf_api_test)
    add_test(NAME api COMMAND npf_api_test)
    add_test(NAME api_runtime1 COMMAND npf_api_test)
    set_tests_properties(api_runtime1 PROPERTIES ENVIRONMENT NPF_TEST_RUNTIME=1)
    add_test(NAME api_runtime2 COMMAND npf_api_test)
    set_tests_properties(api_runtime2 PROPERTIES ENVIRONMENT NPF_TEST_RUNTIME=2)
    add_executable(npf_limits_test test_limits.c)
    target_link_libraries(npf_limits_test PRIVATE npf_shared Threads::Threads)
    npf_target_options(npf_limits_test)
    add_test(NAME limits COMMAND npf_limits_test)
    # npf.h's objects: peers, forwards, resources, events, saving, npf_setup
    add_executable(npf_objects_test test_objects.c)
    target_link_libraries(npf_objects_test PRIVATE npf_shared Threads::Threads)
    npf_target_options(npf_objects_test)
    add_test(NAME objects COMMAND npf_objects_test)
  endif()
  # the contracts: the initialisers' sizes, changes that cannot be saved
  add_executable(npf_contract_test test_contract.c)
  target_link_libraries(npf_contract_test PRIVATE npf_shared Threads::Threads)
  npf_target_options(npf_contract_test)
  add_test(NAME contract COMMAND npf_contract_test)
  add_executable(npf_lifecycle_test test_lifecycle.c)
  target_link_libraries(npf_lifecycle_test PRIVATE npf_common Threads::Threads)
  npf_target_options(npf_lifecycle_test)
  add_test(NAME lifecycle COMMAND npf_lifecycle_test)
  # a program's module (npf/module.h): its own code on the internals
  add_executable(npf_module_test test_module.c)
  target_link_libraries(npf_module_test PRIVATE npf_common Threads::Threads)
  npf_target_options(npf_module_test)
  add_test(NAME module COMMAND npf_module_test)
endif()
